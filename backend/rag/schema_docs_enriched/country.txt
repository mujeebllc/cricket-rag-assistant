
CREATE TABLE country (
	`CountryName` VARCHAR(50), 
	`BoardName` VARCHAR(60), 
	`ISOCode2` VARCHAR(2), 
	`ISOCode3` VARCHAR(3), 
	`CountryCode` INTEGER(11) NOT NULL DEFAULT 92, 
	PRIMARY KEY (`CountryCode`)
)DEFAULT CHARSET=latin1 ENGINE=InnoDB COLLATE latin1_swedish_ci
