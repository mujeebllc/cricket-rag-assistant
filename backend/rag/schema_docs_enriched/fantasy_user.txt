
CREATE TABLE fantasy_user (
	`FUserId` INTEGER(11) NOT NULL AUTO_INCREMENT, 
	`PhoneNo` VARCHAR(15) NOT NULL, 
	`FPassword` VARCHAR(20) NOT NULL, 
	`FullName` VARCHAR(50) NOT NULL, 
	`Address` VARCHAR(80) NOT NULL, 
	`DistrictId` INTEGER(11) NOT NULL, 
	`Gender` ENUM('M','F') NOT NULL, 
	`Email` VARCHAR(50) NOT NULL, 
	`AgeGroup` VARCHAR(12) NOT NULL, 
	`Incomplete` INTEGER(1) NOT NULL DEFAULT 0, 
	`FBId` VARCHAR(30) DEFAULT '0', 
	PRIMARY KEY (`FUserId`)
)DEFAULT CHARSET=latin1 ENGINE=InnoDB COLLATE latin1_swedish_ci
