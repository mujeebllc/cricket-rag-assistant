
CREATE TABLE bowling_detail (
	`MatchNo` INTEGER(11) NOT NULL, 
	`Innings` INTEGER(11) NOT NULL, 
	`PlayerId` INTEGER(11) NOT NULL, 
	`Overs` DOUBLE DEFAULT 0, 
	`Maiden` INTEGER(11) DEFAULT 0, 
	`Runs` INTEGER(11) DEFAULT 0, 
	`Wickets` INTEGER(11) DEFAULT 0, 
	`Wides` INTEGER(11) DEFAULT 0, 
	`NoBalls` INTEGER(11) DEFAULT 0, 
	`TeamId` INTEGER(11), 
	`TeamName` VARCHAR(45) NOT NULL, 
	`Position` INTEGER(11) DEFAULT 0, 
	`Current` INTEGER(11) NOT NULL, 
	`UpdateBy` INTEGER(11) NOT NULL, 
	`LastUpdate` DATETIME NOT NULL, 
	`BowlerName` VARCHAR(100), 
	`MatchType` VARCHAR(1), 
	`Balls` INTEGER(11), 
	PRIMARY KEY (`MatchNo`, `Innings`, `PlayerId`), 
	CONSTRAINT `FK_bowling_detail_innings` FOREIGN KEY(`MatchNo`, `Innings`) REFERENCES innings (`MatchNo`, `Innings`), 
	CONSTRAINT `FK_bowling_detail_player` FOREIGN KEY(`PlayerId`) REFERENCES player (`PlayerId`)
)DEFAULT CHARSET=latin1 ENGINE=InnoDB COLLATE latin1_swedish_ci
