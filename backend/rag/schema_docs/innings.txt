
CREATE TABLE innings (
	`MatchNo` INTEGER(11) NOT NULL, 
	`Innings` INTEGER(11) NOT NULL, 
	`Score` INTEGER(11), 
	`Overs` DOUBLE, 
	`Byes` INTEGER(11), 
	`LByes` INTEGER(11), 
	`Wides` INTEGER(11), 
	`NoBalls` INTEGER(11), 
	`BattingTeam` INTEGER(11), 
	`BowlingTeam` INTEGER(11), 
	`Wickets` INTEGER(11) DEFAULT 0, 
	`UpdateBy` INTEGER(11) NOT NULL, 
	`UpdateTime` DATETIME NOT NULL, 
	`BattingTeamName` VARCHAR(50), 
	`BowlingTeamName` VARCHAR(50), 
	`MatchType` VARCHAR(1) DEFAULT 'C', 
	`CurrentStrikePlayerId` INTEGER(11), 
	`ballsInCurrentOver` INTEGER(11), 
	`CurrentNonStrikePlayerId` INTEGER(11), 
	`CurrentBowlerPlayerId` INTEGER(11), 
	PRIMARY KEY (`MatchNo`, `Innings`), 
	CONSTRAINT `FK_innings_Match` FOREIGN KEY(`MatchNo`) REFERENCES matches (`MatchNo`)
)DEFAULT CHARSET=latin1 ENGINE=InnoDB COLLATE latin1_swedish_ci
