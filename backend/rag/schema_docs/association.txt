
CREATE TABLE association (
	`AssociationId` INTEGER(10) UNSIGNED NOT NULL AUTO_INCREMENT, 
	`AssociationName` VARCHAR(45) NOT NULL DEFAULT ' ', 
	`President` VARCHAR(45) DEFAULT ' ', 
	`Secretary` VARCHAR(45) DEFAULT ' ', 
	`Valid` TINYINT(1) NOT NULL DEFAULT 1, 
	`City` VARCHAR(45) DEFAULT ' ', 
	`Treasurer` VARCHAR(45) DEFAULT ' ', 
	`RegionId` INTEGER(11) NOT NULL, 
	`PresidentContact` VARCHAR(100), 
	`SecretaryContact` VARCHAR(100), 
	`TreasurerContact` VARCHAR(100), 
	`PresidentPic` INTEGER(1) DEFAULT 0, 
	`SecretaryPic` INTEGER(1) DEFAULT 0, 
	`TreasurerPic` INTEGER(1) DEFAULT 0, 
	`ShortName` VARCHAR(10) NOT NULL, 
	PRIMARY KEY (`AssociationId`)
)DEFAULT CHARSET=latin1 ENGINE=InnoDB COLLATE latin1_swedish_ci
