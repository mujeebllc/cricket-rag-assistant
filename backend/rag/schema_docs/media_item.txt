
CREATE TABLE media_item (
	`ItemId` INTEGER(11) NOT NULL AUTO_INCREMENT, 
	`Dated` DATETIME, 
	`Heading` VARCHAR(100), 
	src VARCHAR(50), 
	`GroupId` INTEGER(11), 
	`Height` INTEGER(11) NOT NULL, 
	`Width` INTEGER(11) NOT NULL, 
	PRIMARY KEY (`ItemId`)
)DEFAULT CHARSET=latin1 ENGINE=InnoDB COLLATE latin1_swedish_ci
