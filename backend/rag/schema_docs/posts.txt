
CREATE TABLE posts (
	`PostId` INTEGER(11) NOT NULL AUTO_INCREMENT, 
	`Dated` DATETIME, 
	`Comments` VARCHAR(200), 
	`URL` VARCHAR(80), 
	`PostImg` VARCHAR(80), 
	`Visible` VARCHAR(1) DEFAULT '1', 
	PRIMARY KEY (`PostId`)
)DEFAULT CHARSET=latin1 ENGINE=InnoDB COLLATE latin1_swedish_ci
