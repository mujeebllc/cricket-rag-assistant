
CREATE TABLE batting_stats (
	`PlayerId` INTEGER(11) NOT NULL, 
	`Season` VARCHAR(10) NOT NULL, 
	`Stage` VARCHAR(20) NOT NULL, 
	`Format` VARCHAR(10) NOT NULL, 
	`ICC` VARCHAR(20), 
	`Matches` INTEGER(10) UNSIGNED DEFAULT 0, 
	`Inn` INTEGER(10) UNSIGNED DEFAULT 0, 
	`NotOut` INTEGER(10) UNSIGNED DEFAULT 0, 
	`Runs` INTEGER(10) UNSIGNED DEFAULT 0, 
	`HS` INTEGER(10) UNSIGNED DEFAULT 0, 
	`Average` DECIMAL(6, 2), 
	`BF` INTEGER(10) UNSIGNED DEFAULT 0, 
	`SR` DECIMAL(6, 2), 
	`Hundreds` INTEGER(10) UNSIGNED DEFAULT 0, 
	`Fifties` INTEGER(10) UNSIGNED DEFAULT 0, 
	`Zeros` INTEGER(10) UNSIGNED DEFAULT 0, 
	`Fours` INTEGER(10) UNSIGNED DEFAULT 0, 
	`Sixes` INTEGER(10) UNSIGNED DEFAULT 0, 
	`Catches` INTEGER(11), 
	`Stumps` INTEGER(11), 
	`PlayerName` VARCHAR(50) NOT NULL, 
	`LastMatch` INTEGER(11) DEFAULT 0, 
	PRIMARY KEY (`PlayerId`, `Season`, `Stage`, `Format`)
)DEFAULT CHARSET=utf8mb4 ENGINE=InnoDB COLLATE utf8mb4_uca1400_ai_ci
