
CREATE TABLE season (
	`Season` VARCHAR(7) NOT NULL, 
	`StartDate` DATE, 
	`EndDate` DATE, 
	`Current` VARCHAR(1) DEFAULT '0', 
	PRIMARY KEY (`Season`)
)DEFAULT CHARSET=latin1 ENGINE=InnoDB COLLATE latin1_swedish_ci
