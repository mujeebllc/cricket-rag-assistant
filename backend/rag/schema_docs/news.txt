
CREATE TABLE news (
	`NewsId` INTEGER(11) NOT NULL AUTO_INCREMENT, 
	`RefId` INTEGER(11), 
	`Dated` DATETIME, 
	`Type` VARCHAR(1) DEFAULT 'C', 
	`Urdu` TINYINT(1), 
	`NewsText` VARCHAR(500), 
	PRIMARY KEY (`NewsId`)
)DEFAULT CHARSET=latin1 ENGINE=InnoDB COLLATE latin1_swedish_ci
