
CREATE TABLE club (
	`ClubId` INTEGER(11) NOT NULL AUTO_INCREMENT, 
	`ClubName` VARCHAR(45) NOT NULL DEFAULT ' ', 
	`President` VARCHAR(45) DEFAULT ' ', 
	`Coach` VARCHAR(45) DEFAULT ' ', 
	`Address` VARCHAR(100) DEFAULT ' ', 
	`Description` VARCHAR(500) DEFAULT ' ', 
	`AssociationId` INTEGER(10) UNSIGNED, 
	`Valid` TINYINT(1) NOT NULL DEFAULT 1, 
	`Location` VARCHAR(25) DEFAULT '33.594880, 73.051201', 
	`Logo` TINYINT(1) DEFAULT 0, 
	`ContactNo` VARCHAR(15) DEFAULT ' ', 
	`Treasurer` VARCHAR(45) DEFAULT ' ', 
	`Secretary` VARCHAR(45) DEFAULT ' ', 
	`Registered` TINYINT(1) NOT NULL DEFAULT 1, 
	`ClubPassword` VARCHAR(20) CHARACTER SET latin1 COLLATE latin1_bin, 
	`Cover` TINYINT(4) NOT NULL, 
	`Email` VARCHAR(60) NOT NULL, 
	`Website` VARCHAR(60) NOT NULL, 
	`FBPage` VARCHAR(60) NOT NULL, 
	`Trusted` TINYINT(1) NOT NULL DEFAULT 0, 
	`PresidentContact` VARCHAR(100), 
	`PresidentPic` INTEGER(1) NOT NULL DEFAULT 0, 
	`SecretaryContact` VARCHAR(100), 
	`SecretaryPic` INTEGER(1) NOT NULL DEFAULT 0, 
	`TreasurerContact` VARCHAR(100), 
	`TreasurerPic` INTEGER(1) NOT NULL DEFAULT 0, 
	`ShortName` VARCHAR(10) NOT NULL, 
	`Captain` INTEGER(11), 
	`CCA` INTEGER(11), 
	countryid INTEGER(11), 
	cityid INTEGER(11), 
	city VARCHAR(50), 
	country VARCHAR(30), 
	PRIMARY KEY (`ClubId`), 
	CONSTRAINT `FK_club_association` FOREIGN KEY(`AssociationId`) REFERENCES association (`AssociationId`)
)ROW_FORMAT=DYNAMIC DEFAULT CHARSET=latin1 ENGINE=InnoDB COLLATE latin1_swedish_ci
