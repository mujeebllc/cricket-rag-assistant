
CREATE TABLE live_matches (
	`MatchNo` INTEGER(11) NOT NULL, 
	`Season` VARCHAR(7), 
	`GroundId` INTEGER(11), 
	`Dated` DATETIME, 
	`Winner` INTEGER(10) UNSIGNED, 
	`ResultDetail` VARCHAR(150), 
	`Overs` INTEGER(11), 
	`Status` VARCHAR(1) DEFAULT 'S', 
	`Team1` INTEGER(10) NOT NULL, 
	`Team2` INTEGER(10) NOT NULL, 
	`Type` VARCHAR(20) NOT NULL DEFAULT 'Friendly Match', 
	`TournamentId` INTEGER(10) UNSIGNED, 
	`RoundId` INTEGER(11) NOT NULL DEFAULT 0, 
	`Format` VARCHAR(20) NOT NULL DEFAULT 'One Day', 
	`Level` VARCHAR(10) DEFAULT ' ', 
	`Live` TINYINT(1) NOT NULL DEFAULT 0, 
	`RunnerUP` INTEGER(10), 
	`Summary` VARCHAR(3500), 
	`ManOfMatch` INTEGER(10), 
	`Toss` INTEGER(10), 
	`ScoreCard` INTEGER(11) NOT NULL, 
	`CurrentInn` INTEGER(11) NOT NULL, 
	`Commentary` INTEGER(11) NOT NULL DEFAULT 0, 
	`Official` INTEGER(11) NOT NULL DEFAULT 0, 
	`GroupId` INTEGER(11), 
	`Stage` VARCHAR(15) NOT NULL DEFAULT 'CF', 
	`Scorer` VARCHAR(45), 
	`Umpires` VARCHAR(80) NOT NULL, 
	`Refree` VARCHAR(45) NOT NULL, 
	`CoverPhoto` VARCHAR(50), 
	`TagLine` VARCHAR(60) NOT NULL, 
	`TournamentGroup` VARCHAR(30), 
	`PredictionStart` INTEGER(11) NOT NULL DEFAULT 0, 
	`UpdateBy` INTEGER(11) NOT NULL, 
	`UpdateTime` DATETIME NOT NULL, 
	`Team1RR` DOUBLE NOT NULL DEFAULT 0, 
	`Team2RR` DOUBLE NOT NULL DEFAULT 0, 
	`ScorerId` INTEGER(11) DEFAULT 0, 
	`LiveType` VARCHAR(1), 
	`ResultType` ENUM('WinLoss','Tie','Draw','No Result','Abandoned','Awarded','Conceded') NOT NULL DEFAULT 'WinLoss', 
	`IncludeNetRR` INTEGER(1) NOT NULL DEFAULT 1, 
	`CountryCode` INTEGER(11), 
	`Club` TINYINT(1) DEFAULT 0, 
	`Team` TINYINT(1) DEFAULT 1, 
	`City` INTEGER(11), 
	`MatchLevel` VARCHAR(20), 
	`ICCRecognised` TINYINT(1) DEFAULT 0, 
	`ICCEvent` TINYINT(1) DEFAULT 0, 
	`EntryStatus` VARCHAR(1) DEFAULT 'S', 
	`Team1Name` VARCHAR(45), 
	`Team2Name` VARCHAR(45), 
	`WinnerName` VARCHAR(45), 
	`RunnerupName` VARCHAR(45), 
	`CountryName` VARCHAR(50), 
	`CityName` VARCHAR(50), 
	`LastUpdated` TIMESTAMP NULL DEFAULT (current_timestamp()), 
	PRIMARY KEY (`MatchNo`)
)DEFAULT CHARSET=latin1 ENGINE=InnoDB COLLATE latin1_swedish_ci
