
CREATE TABLE city (
	`CityId` INTEGER(11) NOT NULL AUTO_INCREMENT, 
	`CityName` VARCHAR(50) NOT NULL, 
	`CountryCode` INTEGER(11), 
	PRIMARY KEY (`CityId`), 
	CONSTRAINT fk_city_country FOREIGN KEY(`CountryCode`) REFERENCES country (`CountryCode`)
)DEFAULT CHARSET=latin1 ENGINE=InnoDB COLLATE latin1_swedish_ci
