
CREATE TABLE team (
	`TeamId` INTEGER(11) NOT NULL AUTO_INCREMENT, 
	`TeamName` VARCHAR(50), 
	`Level` VARCHAR(30), 
	`TournamentId` INTEGER(11), 
	`DistrictId` INTEGER(11), 
	`RegionId` INTEGER(11), 
	`DepartmentId` INTEGER(11), 
	`Valid` INTEGER(11), 
	`Format` VARCHAR(20), 
	`Coach` VARCHAR(30), 
	`Manager` VARCHAR(255), 
	`Captain` INTEGER(11), 
	`Season` VARCHAR(7), 
	`Logo` TINYINT(1) NOT NULL DEFAULT 0, 
	`ProvinceId` INTEGER(11), 
	`ShortName` VARCHAR(10) NOT NULL, 
	`Stage` VARCHAR(15), 
	`ParentId` INTEGER(11) NOT NULL DEFAULT 0, 
	`EditionId` INTEGER(11), 
	`LastUpdated` TIMESTAMP NULL DEFAULT (current_timestamp()), 
	PRIMARY KEY (`TeamId`)
)DEFAULT CHARSET=latin1 ENGINE=InnoDB COLLATE latin1_swedish_ci
