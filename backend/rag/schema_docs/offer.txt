
CREATE TABLE offer (
	`OfferId` INTEGER(10) UNSIGNED NOT NULL AUTO_INCREMENT, 
	`ClubId` INTEGER(11) NOT NULL, 
	`OfferDate` DATETIME NOT NULL, 
	`Ground` VARCHAR(70) NOT NULL, 
	`Remove` TINYINT(1) NOT NULL DEFAULT 0, 
	PRIMARY KEY (`OfferId`), 
	CONSTRAINT `FK_offer_Club` FOREIGN KEY(`ClubId`) REFERENCES club (`ClubId`)
)DEFAULT CHARSET=latin1 ENGINE=InnoDB COLLATE latin1_swedish_ci
