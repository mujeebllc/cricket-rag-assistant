
CREATE TABLE point_table (
	`TournamentId` INTEGER(11) NOT NULL, 
	`Type` VARCHAR(1) NOT NULL, 
	`Groupe` VARCHAR(30) NOT NULL, 
	`Team` INTEGER(11) NOT NULL, 
	`TeamName` VARCHAR(45) NOT NULL, 
	`Played` INTEGER(11) NOT NULL DEFAULT 0, 
	`Won` INTEGER(11) NOT NULL DEFAULT 0, 
	`Lost` INTEGER(11) NOT NULL DEFAULT 0, 
	`Draw` INTEGER(11) NOT NULL DEFAULT 0, 
	`Tie` INTEGER(11) NOT NULL DEFAULT 0, 
	`Abondoned` INTEGER(11) NOT NULL DEFAULT 0, 
	`WicketsLost` INTEGER(11) NOT NULL DEFAULT 0, 
	`WicketsTaken` INTEGER(11) NOT NULL DEFAULT 0, 
	`RunsFor` INTEGER(11) NOT NULL DEFAULT 0, 
	`RunsAgainst` INTEGER(11) NOT NULL DEFAULT 0, 
	`Points` INTEGER(11) NOT NULL DEFAULT 0, 
	`BonusPoints` INTEGER(11) NOT NULL DEFAULT 0, 
	`RunRate` DOUBLE NOT NULL DEFAULT 0, 
	`NetRunRate` DOUBLE NOT NULL DEFAULT 0, 
	PRIMARY KEY (`TournamentId`, `Groupe`, `Team`)
)DEFAULT CHARSET=latin1 ENGINE=InnoDB COLLATE latin1_swedish_ci
