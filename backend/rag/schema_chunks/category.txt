
CREATE TABLE category (
	`CatId` INTEGER(10) UNSIGNED NOT NULL AUTO_INCREMENT, 
	`CatName` VARCHAR(50) NOT NULL, 
	`Valid` TINYINT(1) DEFAULT 1, 
	PRIMARY KEY (`CatId`)
)DEFAULT CHARSET=utf8mb4 ENGINE=InnoDB COLLATE utf8mb4_uca1400_ai_ci
