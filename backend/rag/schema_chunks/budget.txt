
CREATE TABLE budget (
	`Id` INTEGER(11) NOT NULL AUTO_INCREMENT, 
	`BudgetMonth` DATE, 
	`HeadId` INTEGER(11), 
	`Amount` INTEGER(11), 
	PRIMARY KEY (`Id`), 
	CONSTRAINT fk_budget_sectionhead FOREIGN KEY(`HeadId`) REFERENCES section_head (`HeadId`)
)DEFAULT CHARSET=utf8mb3 ENGINE=InnoDB COLLATE utf8mb3_general_ci
